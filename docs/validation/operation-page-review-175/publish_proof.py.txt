from pathlib import Path
import gzip
import hashlib
import json
import shutil

root = Path('/Users/dshills/Development/projects/ELab')
stage = Path(__file__).resolve().parent
proof = root/'docs/validation/operation-page-review-175'
receipt_path = root/'research/ui-audit/first-version/operation-page-review-checkpoint.json'
r = json.loads(receipt_path.read_text())
review = Path(r['review_path'])
archive = Path('/private/tmp/elab-175-archive')
package = json.loads((archive/'report.json').read_text())
assert r['version'] == '0.158.0' and r['build'] == '175' and package['passed']
proof.mkdir()
sha = lambda p: hashlib.sha256(p.read_bytes()).hexdigest()
for path in (review/'Evidence').iterdir():
    if path.is_file():
        shutil.copyfile(path, proof/path.name)
    elif path.name in ['Runtime-acceptance', 'Packaged-paging']:
        shutil.copytree(path, proof/('acceptance' if path.name == 'Runtime-acceptance' else 'paging'))
(proof/'archive').mkdir()
for name, target in [('report.json', 'package-report.json.gz'),
                     ('example-report.json', 'example-report.json.gz'),
                     ('signature-report.json', 'signature-report.json.gz')]:
    (proof/'archive'/target).write_bytes(gzip.compress((archive/name).read_bytes(), mtime=0))
(proof/'preservation.json').write_text(json.dumps({k:v for k,v in r.items() if k.startswith('preserved_')}, indent=2)+'\n')
for name in ['publish_proof.py', 'publish_handoff.py']:
