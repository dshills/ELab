def write(path, value):
    with path.open('x') as out:
        out.write(json.dumps(value, indent=2)+'\n')


def gzcopy(source_path, destination):
    with destination.open('xb') as out:
        out.write(gzip.compress(source_path.read_bytes(), mtime=0))


def verify_preserved(folder, manifest_name='RELEASE_MANIFEST.json'):
    manifest = read(folder/manifest_name)
    actual = members(folder)
    assert actual.pop(manifest_name)['sha256'] == sha(folder/manifest_name)
    assert actual == manifest['files'], str(folder)
    return dict(files_verified=len(actual), manifest_sha256=sha(folder/manifest_name))
