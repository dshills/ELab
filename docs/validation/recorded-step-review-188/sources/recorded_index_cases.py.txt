"""Exact copied public index/step commands; mounted input remains source-scoped."""
import hashlib
import json
import math
import time
from floating_cases import close


def check(native, parity, out):
    def session(target, label):
        target.ask('example.rc', dict(discard_changes=True), True)
        target.ask('simulation.pause', {}, True)
        target.ask('simulation.wait')
        target.ask('verification.settings.set', dict(duration_s=.02, sample_step_s=0,
                   input_mode='charge', initial_voltage_v=0, capacitor_initials=[],
                   inductor_initials=[]), True)
        admitted = target.ask('simulation.verify', dict(duration_s=.02, probe_ids=['v:C1', 'i:R1']), True)
        identity = admitted['last_operation']
        deadline = time.monotonic() + 15
        while True:
            captured = target.ask('operation.inspect', dict(id=identity))
            status = captured['operation']['status']
            assert status not in ('failed', 'canceled'), captured
            if status == 'completed':
                break
            assert time.monotonic() < deadline
            time.sleep(.01)
        target.ask('operation.select', dict(id=identity), True)
        target.ask('recording.view.set', dict(mode='recorded'), True)
        original_file = out / (label + '-recorded-index-baseline.elab')
        target.ask('project.save', dict(path=str(original_file)), True)
        original_sha = hashlib.sha256(original_file.read_bytes()).hexdigest()
        before = target.ask('simulation.wait')
        assert before['recording']['id'] == identity
        metadata = target.ask('operation.inspect', dict(id=identity))['metadata']
        rows = target.ask('operation.samples', dict(id=identity, limit=100))['rows']
        assert len(rows) == 21 and [p['id'] for p in metadata['probes']] == ['v:C1', 'i:R1']
        for row in rows:
            decay = math.exp(-row['time_s'] / .01)
            assert abs(row['values'][0] - 5 * (1 - decay)) < 1e-12
            assert abs(row['values'][1] - .0005 * decay) < 1e-15
        # The native viewport reporter can publish unrelated presentation
        # context. Compare all other state, including the complete viewer,
        # live results/history/document and captured operation metadata.
        ignored = {'view', 'viewport', 'revision', 'command_revision', 'last_actor'}
        preserved = lambda value: {k: v for k, v in value.items() if k not in ignored}
        refusals = []
        for index in (.5, 1.5, 19.75):
            baseline = target.ask('project.inspect')
            target.ask('recording.seek', dict(index=index), True, expect='invalid-params')
            refused = target.ask('project.inspect')
            assert preserved(baseline) == preserved(refused)
            refusals.append(dict(index=index, before=baseline, refused=refused))
        stages = []
        for index in (4, 20, 0):
            baseline = target.ask('project.inspect')
            target.ask('recording.seek', dict(index=index), True)
            accepted = target.ask('project.inspect')
            assert accepted['recording']['sample_index'] == index
            assert accepted['recording']['time_s'] == rows[index]['time_s']
            assert {k: v for k, v in preserved(baseline).items() if k != 'recording'} == {k: v for k, v in preserved(accepted).items() if k != 'recording'}
            stages.append(accepted)
        step_stages = []
        for name, seek, speed, direction, expected_index in (
            ('index-next', dict(index=3), .00390625, 'next', 4),
            ('index-previous', dict(index=3), .00390625, 'previous', 2),
            ('time-next', dict(time_s='3 ms'), .00390625, 'next', 4),
            ('time-previous', dict(time_s='3 ms'), .00390625, 'previous', 2),
            ('speed-next', dict(index=8), .0078125, 'next', 9),
            ('speed-previous', dict(index=8), .00390625, 'previous', 7),
        ):
            target.ask('recording.seek', seek, True)
            target.ask('recording.speed.set', dict(seconds_per_wall_second=speed), True)
            baseline = target.ask('project.inspect')
            target.ask('recording.step', dict(direction=direction), True)
            accepted = target.ask('project.inspect')
            viewer = accepted['recording']
            assert not viewer['playing'] and viewer['sample_index'] == expected_index
            assert viewer['time_s'] == rows[expected_index]['time_s']
            assert viewer['seconds_per_wall_second'] == speed
            assert {k: v for k, v in preserved(baseline).items() if k != 'recording'} == {k: v for k, v in preserved(accepted).items() if k != 'recording'}
            step_stages.append(dict(name=name, seek=seek, speed=speed,
                direction=direction, expected_index=expected_index,
                before=baseline, accepted=accepted))
        step_boundaries = []
        for index, direction in ((0, 'previous'), (20, 'next')):
            target.ask('recording.seek', dict(index=index), True)
            baseline = target.ask('project.inspect')
            target.ask('recording.step', dict(direction=direction), True, expect='unavailable')
            refused = target.ask('project.inspect')
            assert preserved(baseline) == preserved(refused)
            step_boundaries.append(dict(index=index, direction=direction,
                before=baseline, refused=refused))
        assert target.ask('operation.inspect', dict(id=identity))['metadata'] == metadata
        assert target.ask('operation.samples', dict(id=identity, limit=100))['rows'] == rows
        final_file = out / (label + '-recorded-index-final.elab')
        target.ask('project.save', dict(path=str(final_file)), True)
        assert hashlib.sha256(original_file.read_bytes()).hexdigest() == original_sha
        assert final_file.read_bytes() == original_file.read_bytes()
        result = dict(before=before, refusals=refusals, integer_stages=stages,
                      metadata=metadata, rows=rows, source_sha256=original_sha,
                      step_stages=step_stages, step_boundaries=step_boundaries,
                      saved_bytes_unchanged=True)
        (out / (label + '-recorded-index.json')).write_text(json.dumps(result, indent=2) + '\n')
        return result

    actual = session(native, 'native')
    reference = parity.helper.Client(command_preconditions=True)
    try:
        expected = session(reference, 'headless')
        close(reference)
    finally:
        if reference.process.poll() is None:
            reference.close()
    assert actual['rows'] == expected['rows']
    for left, right in zip(actual['integer_stages'], expected['integer_stages']):
        for key in ('components', 'wires', 'nets', 'paused'):
            assert left[key] == right[key], key
        for key in ('scope', 'readings', 'sample_index', 'time_s', 'playing'):
            parity.compare(left['recording'][key], right['recording'][key])
    assert len(actual['step_stages']) == len(expected['step_stages']) == 6
    for left, right in zip(actual['step_stages'], expected['step_stages']):
        for key in ('name', 'seek', 'speed', 'direction', 'expected_index'):
            assert left[key] == right[key], key
        for key in ('components', 'wires', 'nets', 'paused'):
            assert left['accepted'][key] == right['accepted'][key], key
        for key in ('scope', 'readings', 'sample_index', 'time_s', 'playing', 'seconds_per_wall_second'):
            parity.compare(left['accepted']['recording'][key], right['accepted']['recording'][key])
    return dict(passed=True, invalid_index_refusals=6, paired_integer_stages=3,
                paired_step_stages=6, step_boundary_refusals=4,
                independent_rc_rows=42, stored_rows_and_metadata_unchanged=True,
                original_and_final_save_bytes_equal=True,
                scope='Exact copied native/headless public seek/speed/step and boundary commands; native presentation revision/viewport fields excluded explicitly. These cases do not invoke the UI callback or claim new physical typing. Source/mounted regressions separately establish first-action field settlement, refusal/correction, boundary drafting and ownership guards.')
