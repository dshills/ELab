from pathlib import Path
import difflib,hashlib,json
r=Path('/Users/dshills/Development/projects/ELab/build/vellum-public-e58d405/drivers/cogent')
u=r/'build/upstream'; p=r/'native/patches/eventposition';p.mkdir()
before_root=Path('/Users/dshills/Development/projects/ELab/build/.vellum-sdk-e58d405-elab2/drivers/cogent/build/upstream')
names=['system/driver/desktop/event.go']+[str(f.relative_to(u)) for f in sorted((u/'system/driver/desktop').glob('event_position_vellum*'))]
inputs={};outputs={};diff=[]
for name in names:
 src=before_root/name;before=src.read_text() if src.exists() else '';after=(u/name).read_text()
 inputs[name]=hashlib.sha256(src.read_bytes()).hexdigest() if src.exists() else None
 outputs[name]=hashlib.sha256((u/name).read_bytes()).hexdigest()
 diff.extend(difflib.unified_diff(before.splitlines(True),after.splitlines(True),fromfile='a/'+name if src.exists() else '/dev/null',tofile='b/'+name))
f=p/'0107-macos-event-position.patch';f.write_text(''.join(diff))
for name,data in [('original-hashes.json',inputs),('output-hashes.json',outputs)]: (p/name).write_text(json.dumps(data,indent=2)+'\n')
m=r/'native/dependencies.json';data=json.loads(m.read_text());data['dependencies'][0]['stages'].append(dict(origin='ELab macOS callback event position compatibility',inputs='native/patches/eventposition/original-hashes.json',outputs='native/patches/eventposition/output-hashes.json',patches=[dict(file='native/patches/eventposition/'+f.name,sha256=hashlib.sha256(f.read_bytes()).hexdigest())]));m.write_text(json.dumps(data,indent=2)+'\n')
