//go:build offscreen

package main

import (
 "reflect"
 "testing"

 "cogentcore.org/core/events"
 "cogentcore.org/core/events/key"
 "github.com/dshills/elab/workbench"
 "github.com/dshills/vellum/drivers/cogent"
)

func TestMountedButtonOwnerGeometrySettlesValueDraftReview(t *testing.T) {
 for _, control := range []string{"rotate-part", "duplicate-part"} {
  t.Run(control,func(t *testing.T) {
   s,cleanup:=testScreen(t);t.Cleanup(cleanup)
   uiAction(t,s,"example.reset",nil)
   agentAction(t,s.service,"simulation.pause",struct{}{})
   agentAction(t,s.service,"component.select",workbench.IDParams{ID:"R1"})
   m:=mountedValuePreset(t,s,control)
   before:=s.service.Snapshot()
   m.input(cogent.ManualInput{Type:events.KeyChord,Code:key.CodeA,Modifiers:key.Meta})
   for _,r:=range "2.2k" { m.input(cogent.ManualInput{Type:events.KeyChord,Rune:r}) }
   if f:=mountedControlElement(t,m,"value-entry");f.Value!="2.2k" || !reflect.DeepEqual(before,s.service.Snapshot()) {t.Fatal("private fixture failed",f.Value)}
   clickMountedControl(t,m,control)
   after:=s.service.Snapshot()
   if s.errorMessage!="" {t.Fatalf("one geometry click refused its own value draft: %s",s.errorMessage)}
   if after.CommandRevision!=before.CommandRevision+2 {t.Fatalf("want draft and one action: %d→%d",before.CommandRevision,after.CommandRevision)}
   if control=="duplicate-part" {
    if len(after.Components)!=len(before.Components)+1 || after.UndoName!="Duplicate component" {t.Fatal("copy did not complete")}
   } else {
    for _,p:=range after.Components {if p.ID=="R1" && p.Rotation!=90 {t.Fatal("rotation did not complete")}}
   }
  })
 }
}
