from pathlib import Path
import hashlib,json,os,shutil,subprocess,time

shared=Path.home()/'Library/Cogent Core'
def geometry():
 assert not any(p.is_symlink() for p in shared.rglob('*'))
 return {p.relative_to(shared).as_posix():dict(sha256=sha(p),mode=oct(p.stat().st_mode&0o777)) for p in shared.rglob('*') if p.is_file()}

(stage/'native-build-info.txt').write_bytes(subprocess.check_output(['go','version','-m',str(binary)]))

try:
 with (stage/'native-smoke-output.txt').open('xb') as output:
  child=subprocess.Popen(argv,cwd=runtime,env=env,stdout=output,stderr=subprocess.STDOUT);pid=child.pid

   child.terminate();code=child.wait(timeout=10);raise

 report=dict(passed=code==0 and failure is None,exit_code=code,pid=pid,argv=argv,elapsed_seconds=time.monotonic()-start,binary_sha256=sha(binary),runtime_binary_sha256=sha(runtime/'elab-native'),endpoint_absent=not endpoint.exists(),geometry_before=before,geometry_after=geometry(),failure=failure,scope='Development modified-source native general smoke with public endpoint, private smoke recovery/appearance storage and shared geometry/repository/network denial.  assert report['shared_geometry_unchanged'] and report['source_unchanged']
print(json.dumps({k:v for k,v in report.items() if k not in ('argv','geometry_before','geometry_after')}))
