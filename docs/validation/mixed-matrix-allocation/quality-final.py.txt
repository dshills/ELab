from pathlib import Path
import hashlib
import json
import os
import subprocess
import time

root = Path('/Users/dshills/Development/projects/ELab')
stage = Path(__file__).resolve().parent
files = sorted(str(p.relative_to(root)) for directory in ('cmd', 'workbench', 'internal', 'transport', 'desktop')
               for p in (root/directory).rglob('*.go'))
sha = lambda p: hashlib.sha256(p.read_bytes()).hexdigest()
before = {name: sha(root/name) for name in files}
start = time.monotonic()
with (stage/'quality-final-default-stdout.txt').open('xb') as stdout, (stage/'quality-final-default-stderr.txt').open('xb') as stderr:
    result = subprocess.run(['./scripts/check.sh'], cwd=root, stdout=stdout, stderr=stderr, timeout=900)
report = dict(passed=result.returncode == 0, exit_code=result.returncode, elapsed_s=time.monotonic()-start,
              command='./scripts/check.sh', source_sha256=before,
              environment={},
              sources_unchanged=before == {name: sha(root/name) for name in files},
              quality_runner_sha256=sha(root/'scripts/check.sh'))
(stage/'quality-final-default.json').write_text(json.dumps(report, indent=2)+'\n')
print({key: report[key] for key in ('passed', 'exit_code', 'elapsed_s', 'sources_unchanged')})
raise SystemExit(result.returncode)
