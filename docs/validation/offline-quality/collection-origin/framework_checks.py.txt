failed=threading.Event();results=[]
def run(name,cmd):
 start=time.monotonic()
 with (base/name).open('w') as output:
  p=subprocess.Popen(cmd,cwd=driver,env=env,stdout=output,stderr=subprocess.STDOUT,start_new_session=True)
  with (base/name).open() as reader:
   tail=''
   while p.poll() is None:
    tail=(tail+reader.read())[-65536:]
    if '"Action":"fail"' in tail:failed.set()
    if failed.is_set():
     try:os.killpg(p.pid,signal.SIGTERM)
     except ProcessLookupError:pass
     break
    time.sleep(.25)
  rc=p.wait()
 row=dict(log=name,command=cmd,exit_code=rc,wall_s=time.monotonic()-start);print(json.dumps(row),flush=True)
 return row
