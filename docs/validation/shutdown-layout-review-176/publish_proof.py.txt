from pathlib import Path
import gzip
import hashlib
import json
import shutil

root=Path('/Users/dshills/Development/projects/ELab')
stage=Path(__file__).resolve().parent
proof=root/'docs/validation/shutdown-layout-review-176'
receipt_path=root/'research/ui-audit/first-version/shutdown-layout-review-checkpoint.json'

sha=lambda p:hashlib.sha256(p.read_bytes()).hexdigest()
for path in (review/'Evidence').iterdir():
    if path.is_file():shutil.copyfile(path,proof/path.name)
    elif path.name in ['Runtime-acceptance','Packaged-paging']:
        shutil.copytree(path,proof/('acceptance' if path.name=='Runtime-acceptance' else 'paging'))
    elif path.name=='Native-shutdown':
        target=proof/'native';target.mkdir()
        refs={}
        for item in path.iterdir():
            assert item.is_file()
            if item.name=='transcript.jsonl.gz':
                refs['transcript']=dict(path=item.relative_to(root).as_posix(),bytes=item.stat().st_size,sha256=sha(item))
            else:shutil.copyfile(item,target/item.name)
        (target/'raw-evidence.json').write_text(json.dumps(refs,indent=2)+'\n')
(proof/'archive').mkdir()
for source,target in [('report.json','package-report.json.gz'),('example-report.json','example-report.json.gz'),('signature-report.json','signature-report.json.gz')]:
    (proof/'archive'/target).write_bytes(gzip.compress((archive/source).read_bytes(),mtime=0))

for name in ['publish_proof.py','publish_handoff.py']:
    shutil.copyfile(stage/name,proof/(name+'.txt'))

native,public=[row['elapsed_seconds'] for row in r['copied_runtime']]
