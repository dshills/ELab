def close(client,label):
    state=client.ask('app.close',dict(discard_changes=True),True);work=state['shutdown']
    assert state['closed'] and work['status']=='completed' and all(work[k] for k in ('workers_joined','recovery_finalized','exit_ready'))
    client.close();assert client.process.returncode==0
    exits.append(dict(session=label,exit_code=0,shutdown=work))

def wait(client,identifier):
    deadline=time.monotonic()+30
    while True:
        record=client.ask('operation.inspect',dict(id=identifier))
        status=record['operation']['status']
        if status=='completed':return record
        assert status in ('queued','running','canceling'),record
        assert time.monotonic()<deadline,'Verify observation deadline'
        time.sleep(.02)
