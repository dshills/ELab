package main
import ("encoding/json";"fmt";"os";"path/filepath";"syscall")
func main(){
 root:="/Users/dshills/Development/projects/ELab"
 executable,err:=os.Executable();if err!=nil{panic(err)}
 run:=filepath.Join(root,"build","opamp-app",fmt.Sprintf("preview-session-%d",os.Getpid()));if err:=os.Mkdir(run,0700);err!=nil{panic(err)}
 endpoint:=filepath.Join(run,"endpoint.json")
 data,_:=json.Marshal(map[string]any{"pid":os.Getpid(),"endpoint":endpoint,"session_dir":run});if err:=os.WriteFile(filepath.Join(root,"build","opamp-app","preview-current.json"),data,0600);err!=nil{panic(err)}
 log,err:=os.OpenFile(filepath.Join(run,"host.log"),os.O_WRONLY|os.O_CREATE|os.O_EXCL,0600);if err!=nil{panic(err)}
 if err:=syscall.Dup2(int(log.Fd()),1);err!=nil{panic(err)};if err:=syscall.Dup2(int(log.Fd()),2);err!=nil{panic(err)}
 host:=filepath.Join(filepath.Dir(executable),"ELab-host")
 if err:=syscall.Exec(host,[]string{host,"--no-recovery","--no-appearance","--agent-endpoint",endpoint},os.Environ());err!=nil{panic(err)}
}
