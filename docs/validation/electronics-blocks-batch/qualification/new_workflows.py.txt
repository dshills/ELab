def wait(read,done,seconds=45):
 end=time.monotonic()+seconds
 while True:
  value=read()
  if done(value):return value
  if time.monotonic()>end:raise TimeoutError('owned feature deadline')
  time.sleep(.02)
def load(c,name):
 c.ask('example.load',dict(id=name,discard_changes=True),mutate=True)
 c.ask('simulation.pause',mutate=True);c.ask('simulation.restart',mutate=True)
 value=c.ask('simulation.wait');assert value['paused'] and not value['diagnostics'],(name,value['diagnostics']);return value
def part(state,name):return next(p for p in state['components'] if p['id']==name)
def capture(c,out,name,duration,step,probes):
 value=c.ask('simulation.verify',dict(duration_s=duration,sample_step_s=step,probe_ids=probes),mutate=True);identity=value['last_operation']
 op=wait(lambda:c.ask('operation.inspect',dict(id=identity)),lambda s:s['operation']['status'] in ('completed','failed','canceled'))
 assert op['operation']['status']=='completed',op
 c.ask('verification.export',dict(id=identity,path=str(out/(name+'.json')),format='json'),mutate=True)
 return identity

def save_reopen(c,out,name):
 before=c.ask('project.inspect');path=out/(name+'.elab');c.ask('project.save',dict(path=str(path)),mutate=True);c.ask('project.load',dict(path=str(path),discard_changes=True),mutate=True);after=c.ask('simulation.wait')
 assert before['components']==after['components'] and before['wires']==after['wires']
 assert after['paused'] and not after['document']['dirty'];return after

 return dict(passed=True,component_kinds=61,commands=220,examples=87,shared_new_commands=commands,block_reused_between_documents=True,thermal_reference_c=thermal['temperature_c'],noise_rms_v=result['noise']['output_rms_v'][0],light_charge_preserved=True,scr_latch_release_reset=True,triac_capture_completed=True,physical_input_acceptance=False)

def main():
