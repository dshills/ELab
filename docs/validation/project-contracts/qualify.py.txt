from pathlib import Path
import datetime,hashlib,json,os,subprocess,time,sys
root=Path.cwd();out=root/'build/project-contracts'
def digest(p):return hashlib.sha256(p.read_bytes()).hexdigest()
public='public' in sys.argv
if not public:
 files=[]
 for base in ['cmd','desktop','internal','workbench','transport','scripts','tools']:
  files.extend(p for p in (root/base).rglob('*') if p.is_file() and '__pycache__' not in p.parts and p.suffix!='.pyc')
 files.extend(root/p for p in ['go.mod','go.sum','third_party/vellum/sdk.lock.json'])
 frozen={str(p.relative_to(root)):digest(p) for p in sorted(set(files)) if p.is_file()}
 (out/'source-freeze.json').write_text(json.dumps(frozen,indent=2)+'\n')
else:frozen=json.loads((out/'source-freeze.json').read_text())
def stable():
 changed=[p for p,h in frozen.items() if digest(root/p)!=h];assert not changed,changed
stages=[]
env=dict(os.environ,GOPROXY='off',GOWORK='off',GOFLAGS='-p=1',GOCACHE=str(root/'build/go-cache'))
commands=[('build',['./scripts/build.sh'],300),('quality',['./scripts/check.sh'],900)]
if public:
 candidate=root/'build/ELab-0.109.0-124-project-contract-candidate.app/Contents/MacOS/ELab'
 commands=[('native-smoke',[str(candidate),'--smoke','--smoke-timeout','2m'],180),('public-suites',[str(root/'build/elab-conformance'),'--native-binary',str(candidate),'--project-contract-suite','--electrical-suite','--inductor-suite','--opamp-suite','--opamp-storage-suite','--capacity-suite','--artifacts',str(out/'public-final2')],900)]
# The runtime candidate/agent hashes are unchanged after the harness-only fix;
# retain its already-passed 45-stage smoke instead of repeating it.
if public:commands=commands[1:]
for name,argv,timeout in commands:
 stable();start=time.monotonic();record={'name':name,'argv':argv,'started_utc':datetime.datetime.now(datetime.timezone.utc).isoformat(),'status':'running'};stages.append(record)
 process=out/('process-public-final.json' if public else 'process-quality-final.json');process.write_text(json.dumps(stages,indent=2)+'\n');print('Starting '+name,flush=True)
 with (out/(name+'-final.log.txt')).open('x') as log:completed=subprocess.run(argv,env=env,stdout=log,stderr=subprocess.STDOUT,timeout=timeout)
 record.update(status='completed' if completed.returncode==0 else 'failed',exit_code=completed.returncode,elapsed_seconds=time.monotonic()-start,finished_utc=datetime.datetime.now(datetime.timezone.utc).isoformat());process.write_text(json.dumps(stages,indent=2)+'\n');stable();print(json.dumps(record),flush=True)
 if completed.returncode:sys.exit(completed.returncode)
