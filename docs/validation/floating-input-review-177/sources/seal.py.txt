def read(path):
    return json.loads(path.read_text())


def sha(path):
    return hashlib.sha256(path.read_bytes()).hexdigest()


def write(path, value):
    with path.open('x') as stream:
        stream.write(json.dumps(value, indent=2)+'\n')


def gzcopy(source, target):
    with target.open('xb') as stream:
        stream.write(gzip.compress(source.read_bytes(), mtime=0))


def preserved(folder, manifest_name='RELEASE_MANIFEST.json'):
    manifest = read(folder/manifest_name)
    actual = members(folder)
    actual.pop(manifest_name)
    assert actual == manifest['files'], folder
    return dict(files_verified=len(actual), manifest_sha256=sha(folder/manifest_name))
